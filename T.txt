    <script>
        // -------------------------------------------------------------
        // CLOUD COMMUNICATION ENGINE (Double Cloud Sync - High Compatibility)
        // -------------------------------------------------------------
        
        // 採用多重公共 REST Gateway 避免單點網絡阻擋
        const ENDPOINTS = [
            "https://api.rest7.com/v1/json/push",
            "https://hk-classroom-quiz-default-rtdb.asia-southeast1.firebasedatabase.app"
        ];

        // 升級版：連線發送函式（支援連續握手發送）
        async function pushDataToCloud() {
            if (!state.roomCode) return;
            const dataPayload = JSON.stringify(state.groups);

            // 1. 本地廣播 (若剛好在同一台機器/分頁)
            try {
                if (window.BroadcastChannel) {
                    const bc = new BroadcastChannel('quiz_channel_' + state.roomCode);
                    bc.postMessage(state.groups);
                }
            } catch(e){}

            // 2. 雲端雙向同步 (HTTP REST Long-Poll Gateway)
            const url = `https://hk-classroom-quiz-default-rtdb.asia-southeast1.firebasedatabase.app/rooms/${state.roomCode}.json`;

            try {
                await fetch(url, {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: dataPayload,
                    mode: 'cors'
                });
                if (state.role === 'student') {
                    showSyncStatus("已發送至老師大螢幕！");
                }
            } catch (err) {
                console.warn("Primary endpoint retry...", err);
                // 備用快取備份
                localStorage.setItem('room_' + state.roomCode, dataPayload);
            }
        }

        // 輪詢與即時監聽 (老師端與同學端通用)
        function startPollingServer() {
            // 本地頻道監聽
            try {
                if (window.BroadcastChannel) {
                    const bc = new BroadcastChannel('quiz_channel_' + state.roomCode);
                    bc.onmessage = (event) => {
                        if (event.data) mergeCloudData(event.data);
                    };
                }
            } catch(e){}

            // 每 800ms 超高速實時輪詢
            setInterval(async () => {
                if (!state.roomCode) return;
                const url = `https://hk-classroom-quiz-default-rtdb.asia-southeast1.firebasedatabase.app/rooms/${state.roomCode}.json`;

                try {
                    const res = await fetch(url, { cache: 'no-cache' });
                    if (res.ok) {
                        const data = await res.json();
                        if (data) {
                            mergeCloudData(data);
                        }
                    }
                } catch (err) {
                    // 若網絡中斷，嘗試讀取 LocalStorage 備份
                    const localData = localStorage.getItem('room_' + state.roomCode);
                    if (localData) mergeCloudData(JSON.parse(localData));
                }
            }, 800);
        }

        function startStudentSession() {
            const roomCode = document.getElementById('inputRoomCode').value.trim();
            const groupNum = document.getElementById('selectGroup').value;
            const name = document.getElementById('inputStudentName').value.trim();

            if (!roomCode || roomCode.length !== 4) {
                alert("請輸入正確的 4 位數房間號碼！");
                return;
            }
            if (!groupNum || !name) {
                alert("請完整選擇組別並輸入隊員姓名！");
                return;
            }

            state.roomCode = roomCode;
            state.studentGroup = groupNum;
            state.studentName = name;

            document.getElementById('roomNumDisplay').innerText = roomCode;
            document.getElementById('studentSetupModal').classList.add('hidden');
            document.getElementById('studentFrame').classList.remove('hidden');
            document.getElementById('studentFrame').classList.add('flex');

            document.getElementById('studentGroupBadge').innerText = `第 ${groupNum} 組`;
            document.getElementById('studentNameDisplay').innerText = name;

            // 1. 本地記錄
            state.groups[groupNum].name = name;

            // 2. ⚡ 強制連續發送 3 次連線信號，確保老師端一定能收到「已登場」
            pushDataToCloud();
            setTimeout(pushDataToCloud, 500);
            setTimeout(pushDataToCloud, 1200);

            // 3. 啟動同學端實時同步
            startPollingServer();
        }

        function handleStudentInput() {
            const val = document.getElementById('studentAnswerInput').value;
            document.getElementById('charCounter').innerText = `${val.length} 字`;

            state.groups[state.studentGroup].answer = val;
            state.groups[state.studentGroup].name = state.studentName;

            // Debounce 發送 (避免每按一個字都極速 Call API)
            clearTimeout(state.syncDebounceTimer);
            state.syncDebounceTimer = setTimeout(() => {
                pushDataToCloud();
            }, 300);
        }

        // -------------------------------------------------------------
        // CLOUD COMMUNICATION ENGINE (Firebase REST API / HTTPS)
        // -------------------------------------------------------------
        // 發送資料至雲端
        async function pushDataToCloud() {
            if (!state.roomCode) return;
            const url = `${DB_BASE_URL}/rooms/${state.roomCode}.json`;

            try {
                await fetch(url, {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(state.groups)
                });
                if (state.role === 'student') {
                    showSyncStatus("已同步");
                }
            } catch (err) {
                console.error("Cloud push failed:", err);
            }
        }

        // 輪詢從雲端拉取最新資料 (1 秒一次)
        function startPollingServer() {
            setInterval(async () => {
                if (!state.roomCode) return;
                const url = `${DB_BASE_URL}/rooms/${state.roomCode}.json`;

                try {
                    const res = await fetch(url);
                    if (res.ok) {
                        const data = await res.json();
                        if (data) {
                            mergeCloudData(data);
                        }
                    }
                } catch (err) {
                    console.error("Cloud poll failed:", err);
                }
            }, 1000);
        }

        function mergeCloudData(cloudGroups) {
            for (let i = 1; i <= 8; i++) {
                if (!cloudGroups[i]) continue;

                // 如果是同學端，保留自己的答案輸入內容，只同步老師評分與貼紙
                if (state.role === 'student') {
                    if (i == state.studentGroup) {
                        state.groups[i].score = cloudGroups[i].score || 0;
                        state.groups[i].stickers = cloudGroups[i].stickers || [];
                        continue; 
                    }
                }

                // 老師端或同學端其他組別：完全更新數據
                state.groups[i] = {
                    name: cloudGroups[i].name || '',
                    answer: cloudGroups[i].answer || '',
                    score: cloudGroups[i].score || 0,
                    stickers: cloudGroups[i].stickers || []
                };

                updateGroupUI(i);
            }
        }

        function updateGroupUI(gNum) {
            if (state.role !== 'teacher') return;

            const gData = state.groups[gNum] || { name: '', answer: '', score: 0, stickers: [] };

            // 隊員姓名
            const nameEl = document.getElementById(`groupName${gNum}`);
            if (nameEl) {
                if (gData.name) {
                    nameEl.innerHTML = `<span class="inline-block w-2 h-2 rounded-full bg-emerald-500 mr-1 animate-ping"></span>(${gData.name})`;
                    nameEl.className = "text-xs md:text-sm font-bold text-sky-700 truncate flex items-center";
                } else {
                    nameEl.innerText = '(未登入)';
                    nameEl.className = "text-xs md:text-sm font-bold text-slate-400 truncate";
                }
            }

            // 答案內容
            const ansEl = document.getElementById(`groupAnswer${gNum}`);
            if (ansEl) {
                if (gData.answer && gData.answer.trim()) {
                    ansEl.innerText = gData.answer;
                    ansEl.className = "text-slate-800 font-bold text-sm md:text-base leading-relaxed whitespace-pre-wrap";
                } else {
                    ansEl.innerHTML = `<span class="text-slate-300 font-normal italic">等待組別輸入答案...</span>`;
                }
            }

            // 星級評分
            updateStarDisplay(gNum, gData.score || 0);

            // 貼紙
            renderStickers(gNum);
        }

        function exportResults() {
            let summary = "=== 課堂問答大賽 - 最終得分與答題紀錄 ===\n\n";
            summary += `日期：${new Date().toLocaleString('zh-HK')}\n`;
            summary += `房間號碼：${state.roomCode}\n`;
            summary += "--------------------------------------------------\n\n";

            for (let i = 1; i <= 8; i++) {
                const g = state.groups[i] || { name: "未填寫", answer: "無", score: 0, stickers: [] };
                summary += `【第 ${i} 組】\n`;
                summary += `參賽隊員：${g.name || "未填寫"}\n`;
                summary += `評分：${'★'.repeat(g.score || 0)}${'☆'.repeat(5 - (g.score || 0))} (${g.score || 0}/5 星)\n`;
                summary += `獲頒貼紙：${(g.stickers && g.stickers.length > 0) ? g.stickers.join(', ') : '無'}\n`;
                summary += `作答答案：\n${g.answer || "無內容"}\n`;
                summary += "--------------------------------------------------\n";
            }

            const blob = new Blob([summary], { type: 'text/plain;charset=utf-8' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `課堂問答大賽總結報告_${new Date().toISOString().slice(0,10)}.txt`;
            link.click();
        }
